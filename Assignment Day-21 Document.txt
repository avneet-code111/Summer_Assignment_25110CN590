
//Q-81) Write a program to Find string length without strlen(). 

#include <stdio.h>

int main() {
    char str[100];
    int i = 0;

    scanf("%s", str);

    while(str[i]!= '\0') {
        i = i + 1;
    }

    printf("%d", i);

    return 0;
}

///Q-82) Write a program to Reverse a string. 

#include <stdio.h>

int main() {
    char str[100];
    int i, len = 0;

    scanf("%s", str);

    while(str[len]!= '\0') {
        len = len + 1;
    }

    for(i = len - 1; i >= 0; i--) {
        printf("%c", str[i]);
    }

    return 0;
}

///Q-83) Write a program to Count vowels and consonants. 

#include <stdio.h>

int main() {
    char str[100];
    int i = 0, vowels = 0, consonants = 0;

    scanf("%s", str);

    while(str[i]!= '\0') {
        if(str[i] == 'a' || str[i] == 'e' || str[i] == 'i' || str[i] == 'o' || str[i] == 'u') {
            vowels = vowels + 1;
        }
        else if(str[i] == 'A' || str[i] == 'E' || str[i] == 'I' || str[i] == 'O' || str[i] == 'U') {
            vowels = vowels + 1;
        }
        else if((str[i] >= 'a' && str[i] <= 'z') || (str[i] >= 'A' && str[i] <= 'Z')) {
            consonants = consonants + 1;
        }
        i = i + 1;
    }

    printf("%d\n", vowels);
    printf("%d", consonants);

    return 0;
}

///Q-84) Write a program to Convert lowercase to uppercase. 

#include <stdio.h>

int main() {
    char ch;

    scanf("%c", &ch);

    if(ch >= 'a' && ch <= 'z') {
        ch = ch - 32;
    }

    printf("%c", ch);

    return 0;
}
